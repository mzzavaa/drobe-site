{{- /* One stylesheet. Concatenated so the page makes a single CSS request, and fingerprinted so a change is never served from a stale cache. */ -}} {{- $tokens := resources.Get "css/tokens.css" -}} {{- $base := resources.Get "css/base.css" -}} {{- $comp := resources.Get "css/components.css" -}} {{- $css := slice $tokens $base $comp | resources.Concat "css/drobe.css" -}} {{- if hugo.IsProduction -}} {{- $css = $css | minify | fingerprint -}} {{- end -}}
